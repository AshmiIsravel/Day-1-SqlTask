-- 3. Create Books table
CREATE TABLE Books (
    book_id INT PRIMARY KEY,
    book_name VARCHAR(150),
    author VARCHAR(100),
    price DECIMAL(10,2)
);


-- 4. Create Members table
CREATE TABLE Members (
    member_id INT PRIMARY KEY,
    member_name VARCHAR(100),
    city VARCHAR(100),
    phone VARCHAR(20)
);


-- 5. Create Borrow table
CREATE TABLE Borrow (
    borrow_id INT PRIMARY KEY,
    book_id INT,
    member_id INT,
    borrow_date DATE
);

ALTER TABLE Books

============================================================
A4 - DCL - DATA CONTROL LANGUAGE
============================================================

-- 1. Create user

CREATE USER library_user
WITH PASSWORD 'Library@123';


-- 2. Grant SELECT privilege

GRANT SELECT
ON TABLE Books
TO library_user;


-- 3. Grant INSERT privilege

GRANT INSERT
ON TABLE Books
TO library_user;


-- 4. Grant UPDATE privilege

GRANT UPDATE
ON TABLE Books
TO library_user;


-- 5. Display privileges

SELECT grantee, privilege_type
FROM information_schema.role_table_grants
WHERE grantee = 'library_user'
AND table_name = 'books';


-- 6. Revoke INSERT privilege

REVOKE INSERT
ON TABLE Books
FROM library_user;


-- 7. Revoke UPDATE privilege

REVOKE UPDATE
ON TABLE Books
FROM library_user;


-- 8. Grant SELECT on all tables

GRANT SELECT
ON ALL TABLES IN SCHEMA public
TO library_user;


-- 9. Revoke SELECT on Books

REVOKE SELECT
ON TABLE Books
FROM library_user;


-- 10. Display final privileges

SELECT grantee, table_name, privilege_type
FROM information_schema.role_table_grants
WHERE grantee = 'library_user';

FROM Books
GROUP BY category;


-- 68. Lowest price in each category

FROM Books
GROUP BY category;


-- 69. Total stock quantity for each category

SELECT category, SUM(stock_quantity) AS total_stock
FROM Books
GROUP BY category;


-- 70. Total value of books in each category

FROM Books
GROUP BY category;


-- 71. Categories having more than 2 books

SELECT category, COUNT(*) AS book_count
FROM Books
GROUP BY category
HAVING COUNT(*) > 2;


-- 72. Categories whose average price is greater than 500
